import { Bot, InlineKeyboard } from "grammy"; const bot = new Bot("የእርስዎ_ትክክለኛ_የቦት_ቶከን"); // የሚኒ አፑ የሚስተናገድበት ድረ-ገጽ አድራሻ (Cloudflare Pages ወይም ሌላ URL) const MINI_APP_URL = "https://your-driver-app.pages.dev"; // 1. ቦቱ ሲጀመር ሚኒ አፑን የሚከፍት አዝራር መስጠት bot.command("start", async (ctx) => { const keyboard = new InlineKeyboard().webApp( "📝 የምዝገባ ቅጽ ክፈት", MINI_APP_URL ); await ctx.reply( "👋 እንኳን ደህና መጡ ወደ አሽከርካሪዎች መመዝገቢያ ፖርታል!\n\nምዝገባዎን በአንድ ገጽ ላይ በፍጥነት ለማጠናቀቅ ከታች ያለውን አዝራር ይጫኑ፦", { reply_markup: keyboard } ); }); // 2. ከሚኒ አፑ የተላከውን መረጃ መቀበል (tg.sendData) bot.on("message:web_app_data", async (ctx) => { const data = JSON.parse(ctx.message.web_app_data.data); await ctx.reply( `🎉 **የምዝገባ ጥያቄዎ በተሳካ ሁኔታ ደርሶናል!**\n\n` + `👤 **ስም፦** ${data.fullName}\n` + `📞 **ስልክ፦** ${data.phoneNumber}\n` + `🚗 **ተሽከርካሪ፦** ${data.vehicleModel}\n` + `🔢 **ሰሌዳ፦** ${data.plateCode}-${data.plateNumber}\n\n` + `መረጃዎ በአስተዳዳሪው ተገምግሞ ምላሽ ይሰጥዎታል።` ); }); bot.start();
